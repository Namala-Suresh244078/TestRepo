```ts
import { Tabulator, TableExtensions } from 'survey-analytics/survey.analytics.tabulator';

private renderDataTable(form: Model, formResults: NonNullable<unknown>[]) {

  const rowSelectExt = TableExtensions.findExtension('row', 'select');
  const rowDetailsExt = TableExtensions.findExtension('row', 'details');

  if (rowSelectExt) {
    rowSelectExt.visibleIndex = 0;
  }

  if (rowDetailsExt) {
    rowDetailsExt.visibleIndex = 1;
  }

  const removeRowsExt = TableExtensions.findExtension('header', 'removerows');

  if (removeRowsExt) {
    removeRowsExt.visibleIndex = 5;
  }

  const tabulatorConfiguration = {
    tabulatorOptions: {
      selectable: true,
      selectablePersistence: false,
      tooltipsHeader: false,
      columnDefaults: { tooltip: false },
    },
  };

  this.formDataTable = new Tabulator(
    form,
    formResults,
    tabulatorConfiguration
  );

  this.formDataTable.render('form-results-data-table');

  this.tabulatorTables = this.formDataTable.tabulatorTables;

  this.tabulatorTables.on('tableBuilt', () => {
    const table = this.tabulatorTables;

    table.setSort([
      {
        column: 'Submission Date/Time',
        dir: 'desc',
      },
    ]);

    table.on('rowSelectionChanged', () => {
      console.log('selection changed');
    });
  });
}
```
